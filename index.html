// ------------------------------------------------------------- // TEMPORARY ITEMS LOGIC (FOR SELLER) // ------------------------------------------------------------- function addTempItem() { const nameInput = document.getElementById('tempItemName'); const categoryInput = document.getElementById('tempItemCategory'); const priceInput = document.getElementById('tempItemPrice'); const imageInput = document.getElementById('tempItemImage'); if (!nameInput.value.trim() || !priceInput.value) { alert('تکایە ناوی کاڵا و نرخی کاڵاکە بنووسە.'); return; } const name = nameInput.value.trim(); const category = categoryInput.value; const price = parseFloat(priceInput.value); const file = imageInput.files[0]; if (file) { const reader = new FileReader(); reader.onload = function (e) { saveTempItem(name, category, price, e.target.result); }; reader.readAsDataURL(file); } else { saveTempItem(name, category, price, 'https://via.placeholder.com/80/1e293b/f59e0b?text=No+Image'); } } function saveTempItem(name, category, price, imageUrl) { tempSellerItems.push({ id: Date.now(), name, category, price, imageUrl }); // Reset inputs document.getElementById('tempItemName').value = ''; document.getElementById('tempItemPrice').value = ''; document.getElementById('tempItemImage').value = ''; renderTempItemsTable(); } function renderTempItemsTable() { const tbody = document.getElementById('tempItemsTable'); if (tempSellerItems.length === 0) { tbody.innerHTML = ` هیچ کاڵایەک زیادر نەکراوە. `; return; } tbody.innerHTML = tempSellerItems.map((item, index) => ` ${index + 1} ${item.name} ${item.category} $${item.price.toFixed(2)} `).join(''); } function removeTempItem(id) { tempSellerItems = tempSellerItems.filter(item => item.id !== id); renderTempItemsTable(); } // Final Form Submission by Seller function handleFormSubmit(e) { e.preventDefault(); if (tempSellerItems.length === 0) { alert('تکایە لانیکەم کاڵایەک زیابکە بۆ لیستەکە بەرلە ناردنی کۆتایی!'); return; } const formattedItems = tempSellerItems.map(item => ({ ...item, sellerStore: currentSellerSession.storeName, sellerPhone: currentSellerSession.phone, sellerLocation: currentSellerSession.location })); masterProducts = [...masterProducts, ...formattedItems]; localStorage.setItem('official_master_products', JSON.stringify(masterProducts)); tempSellerItems = []; renderTempItemsTable(); alert('✅ سەرکەوتووانە هەموو کاڵاکان تۆمارکران و ڕەوانەی سیستەمی بەراوردکاری کەوتن!'); } // ------------------------------------------------------------- // ADMIN DASHBOARD LOGIC // ------------------------------------------------------------- function switchAdminTab(tab) { const productsSec = document.getElementById('adminProductsSection'); const sellersSec = document.getElementById('adminSellersSection'); const productsBtn = document.getElementById('adminTabProductsBtn'); const sellersBtn = document.getElementById('adminTabSellersBtn'); if (tab === 'products') { productsSec.classList.remove('hidden'); sellersSec.classList.add('hidden'); productsBtn.className = "px-5 py-2.5 bg-slate-900 text-amber-400 rounded-xl text-xs font-bold transition flex items-center gap-2 shadow"; sellersBtn.className = "px-5 py-2.5 bg-white text-slate-700 hover:bg-slate-100 rounded-xl text-xs font-bold transition flex items-center gap-2 border border-slate-200"; } else { sellersSec.classList.remove('hidden'); productsSec.classList.add('hidden'); sellersBtn.className = "px-5 py-2.5 bg-slate-900 text-amber-400 rounded-xl text-xs font-bold transition flex items-center gap-2 shadow"; productsBtn.className = "px-5 py-2.5 bg-white text-slate-700 hover:bg-slate-100 rounded-xl text-xs font-bold transition flex items-center gap-2 border border-slate-200"; } } function renderAdminDashboard() { const searchVal = document.getElementById('adminSearchInput').value.toLowerCase().trim(); const catVal = document.getElementById('adminCategoryFilter').value; let filtered = masterProducts.filter(item => { const matchesSearch = item.name.toLowerCase().includes(searchVal) || item.sellerStore.toLowerCase().includes(searchVal); const matchesCat = (catVal === 'هەمووی') || (item.category === catVal); return matchesSearch && matchesCat; }); // Update Statistics Cards const totalStores = new Set(masterProducts.map(p => p.sellerStore)).size; document.getElementById('statTotalStores').innerText = totalStores; document.getElementById('statTotalItems').innerText = filtered.length; if (filtered.length > 0) { const minItem = filtered.reduce((prev, curr) => prev.price < curr.price ? prev : curr); const maxItem = filtered.reduce((prev, curr) => prev.price > curr.price ? prev : curr); document.getElementById('statMinPrice').innerText = `$${minItem.price.toFixed(2)}`; document.getElementById('statMinDetails').innerText = `${minItem.name} (${minItem.sellerStore})`; document.getElementById('statMaxPrice').innerText = `$${maxItem.price.toFixed(2)}`; document.getElementById('statMaxDetails').innerText = `${maxItem.name} (${maxItem.sellerStore})`; } else { document.getElementById('statMinPrice').innerText = '$0'; document.getElementById('statMinDetails').innerText = 'فرۆشگا: --'; document.getElementById('statMaxPrice').innerText = '$0'; document.getElementById('statMaxDetails').innerText = 'فرۆشگا: --'; } document.getElementById('tableCountBadge').innerText = `${filtered.length} کاڵا`; const tbody = document.getElementById('adminDataTable'); if (filtered.length === 0) { tbody.innerHTML = `هیچ زانیارییەک نەدۆزرایەوە.`; return; } tbody.innerHTML = filtered.map(item => ` ${item.name} ${item.category} ${item.sellerStore} ${item.sellerLocation} ${item.sellerPhone} $${item.price.toFixed(2)} `).join(''); } function deleteMasterProduct(id) { if (confirm('دڵنیایت لە سڕینەوەی ئەم کاڵایە؟')) { masterProducts = masterProducts.filter(p => p.id !== id); localStorage.setItem('official_master_products', JSON.stringify(masterProducts)); renderAdminDashboard(); } } // Seller Management (Block / Unblock) function renderAdminSellers() { document.getElementById('sellerCountBadge').innerText = `${registeredSellers.length} فرۆشگا`; const tbody = document.getElementById('adminSellersTable'); if (registeredSellers.length === 0) { tbody.innerHTML = `هیچ فرۆشیارێک تۆمار نەکراوە.`; return; } tbody.innerHTML = registeredSellers.map(seller => ` ${seller.storeName} ${seller.phone} ${seller.location} ${seller.isBlocked ? `بلۆککراوە` : `چالاک` } `).join(''); } function toggleBlockSeller(phone) { registeredSellers = registeredSellers.map(s => { if (s.phone === phone) { return { ...s, isBlocked: !s.isBlocked }; } return s; }); localStorage.setItem('registered_sellers', JSON.stringify(registeredSellers)); renderAdminSellers(); } // Initialize Application UI State updateAppUI();
